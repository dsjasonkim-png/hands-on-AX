## 4.1 도구 통합하기

### 4.1.2 @tool 데코레이터로 도구 만들기

```py 원고
from langchain_core.tools import tool

@tool(parse_docstring=True)
def lookup_order(order_id: str, include_items: bool = False) -> str:
    """Z몰 주문 DB에서 주문번호로 주문 상태를 조회합니다. 읽기 전용입니다.

    Args:
        order_id: 조회할 주문번호 (예: ZM-1029)
        include_items: 주문 상품 목록을 함께 반환할지 여부 (기본값: False)
    """
    # 실제로는 여기에 DB 조회 코드가 들어갑니다
    return f"주문 {order_id}: 배송 중, 11/09 도착 예정"

# 도구 정보 확인
print(lookup_order.name)
print(lookup_order.description)
print(lookup_order.args)
```

### 4.1.3 도구 이름과 설명 커스터마이징

```py 원고
from langchain_core.tools import tool

@tool("discount_calculator", description="가격과 할인율로 할인 금액을 계산합니다. 할인 계산이 필요할 때만 사용하세요.")
def calculate_discount(price: int, rate: int) -> float:
    """정수 가격과 0~100의 정수 할인율로 할인 금액을 계산합니다."""
    if price < 0 or not 0 <= rate <= 100:
        raise ValueError("가격과 할인율 범위를 확인하세요.")
    return price * rate / 100

print(calculate_discount.name) # discount_calculator
print(calculate_discount.description) # 가격과 할인율로 할인 금액을 계산합니다...
```

### 4.1.4 bind_tools와 도구 호출 루프

```py 원고
from langchain.chat_models import init_chat_model
from langchain_core.messages import SystemMessage, HumanMessage

model = init_chat_model("gpt-4o-mini", temperature=0)
model_with_tools = model.bind_tools([lookup_order])

messages = [
    SystemMessage("당신은 Z몰 고객지원 담당입니다. 조회 결과에 없는 내용은 답하지 않습니다."),
    HumanMessage("ZM-1029 주문 어디쯤이야?"),
]

# 1단계: 모델이 도구 호출을 요청합니다
ai_msg = model_with_tools.invoke(messages)
print(ai_msg.tool_calls)
```

```py 원고
# 2단계: 프로그램이 도구를 실행합니다
messages.append(ai_msg)
for tool_call in ai_msg.tool_calls:
    tool_msg = lookup_order.invoke(tool_call) # ToolMessage가 반환됩니다
    messages.append(tool_msg)

# 3단계: 조회 결과를 받은 모델이 최종 답변을 만듭니다
final = model_with_tools.invoke(messages)
print(final.content)
```

```py 보조
#! note: 대화 기록이 어떻게 쌓였는지 확인합니다.
for i, m in enumerate(messages + [final]):
    print(f"[{i}] {type(m).__name__:13}:", m.content or m.tool_calls)
```

### 4.1.5 복잡한 입력 다루기: Pydantic 활용

```py 원고
from pydantic import BaseModel, Field
from typing import Literal
from langchain_core.tools import tool

class SupplyRequestInput(BaseModel):
    item: str = Field(description="신청할 비품 이름")
    quantity: int = Field(description="신청 수량", ge=1)
    category: Literal["사무용품", "IT기기", "기타"] = Field(
        default="사무용품",
        description="비품 구분 (사무용품, IT기기, 기타 중 하나)"
    )

@tool(args_schema=SupplyRequestInput)
def check_supply_request(item: str, quantity: int, category: str) -> str:
    """비품 신청서의 입력 값을 확인합니다. 조회만 하며 신청을 접수하지 않습니다."""
    return f"{category} 구분 '{item}' {quantity}개 신청서를 확인했습니다."

result = check_supply_request.invoke({
    "item": "모니터 받침대",
    "quantity": 3,
    "category": "사무용품"
})
print(result)
```

### 4.1.6 컨텍스트 접근: 사용자 정보를 안전하게 전달하기

```py 원고
from langchain_core.tools import tool
from langchain_core.runnables import RunnableConfig

@tool
def get_user_profile(config: RunnableConfig) -> str:
    """현재 접속 중인 고객의 프로필 정보를 가져옵니다."""
    # 시스템이 주입한 정보를 꺼냅니다. 모델은 이 값을 직접 제어하지 못합니다.
    user_id = config["configurable"].get("user_id", "unknown")
    username = config["configurable"].get("username", "unknown")
    return f"고객 ID: {user_id}, 이름: {username}"

# 도구 호출 시 config로 사용자 정보를 주입합니다
result = get_user_profile.invoke(
    {},
    config={"configurable": {"user_id": "user_1029", "username": "김민지"}}
)
print(result)
```

```py 보조
#! note: 모델에게 보이는 도구 입력에는 config가 없습니다. 모델은 이 값을 채울 수 없습니다.
print(get_user_profile.args)
```

### 4.1.7 조회 도구와 변경 도구 구분하기

```py 원고
from langchain_core.runnables import RunnableConfig

@tool
def request_refund(order_id: str, reason: str, config: RunnableConfig) -> str:
    """환불 요청서를 만듭니다. 환불을 실행하지 않으며 담당자 승인이 필요합니다."""
    user_id = config["configurable"]["user_id"]
    # 서버에서 확인: 이 고객의 주문인가, 환불 가능 상태인가, 금액은 얼마인가
    # 확인을 통과하면 요청서를 저장하고 승인 대기 상태로 둡니다
    return f"주문 {order_id} 환불 요청서를 만들었습니다. 담당자 승인 후 처리됩니다."
```

## 4.2 구조화된 출력

### 4.2.2 with_structured_output으로 구조화 출력 받기

```py 원고
from langchain.chat_models import init_chat_model
from pydantic import BaseModel, Field

class ContactInfo(BaseModel):
    name: str = Field(description="고객 이름")
    email: str = Field(description="이메일 주소")
    phone: str = Field(description="전화번호")

model = init_chat_model("gpt-4o")
structured_model = model.with_structured_output(ContactInfo)

result = structured_model.invoke(
    "김민지, minji@example.com, 010-1234-5678 정보를 정리해줘"
)
print(result)
print(type(result))
```

```py 원고
class ProductReview(BaseModel):
```

### 4.2.5 형식 검증과 업무 검증

```py 원고
from pydantic import BaseModel, Field

class RefundDecision(BaseModel):
    order_id: str = Field(description="주문번호")
    refund_amount: int = Field(description="환불 금액(원)", ge=0)
    reason: str = Field(description="환불 사유")

decision = RefundDecision(order_id="ZM-1029", refund_amount=54000, reason="단순 변심")
print(decision)
```

```py 원고
def validate_refund(decision: RefundDecision, order: dict) -> list[str]:
    """모델의 환불 판단을 주문 데이터와 대조합니다."""
    errors = []
    if decision.order_id != order["order_id"]:
        errors.append("주문번호가 조회 결과와 다릅니다.")
    max_refund = order["paid_amount"]
    if decision.reason == "단순 변심":
        max_refund -= 5000 # 왕복 배송비 차감 (5장의 반품 정책)
    if decision.refund_amount > max_refund:
        errors.append(f"환불 금액이 한도({max_refund:,}원)를 넘습니다.")
    return errors

order = {"order_id": "ZM-1029", "paid_amount": 49000} # lookup_order 결과
print(validate_refund(decision, order))
```

### 4.2.6 구조화된 출력은 에이전트와 시스템 사이의 계약이다

```py 원고
from typing import Literal
from pydantic import BaseModel, Field

class InquiryDecision(BaseModel):
    status: Literal["완료", "정보 부족", "검토 필요", "실패"]
    category: str = Field(description="문의 유형 (예: 배송, 반품, 결제)")
    reply_draft: str = Field(description="고객에게 보낼 답변 초안")
    evidence: list[str] = Field(description="답변 근거가 된 주문·정책 항목 ID")
```

```py 보조

decision = model.with_structured_output(InquiryDecision).invoke(
```
