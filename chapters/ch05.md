## 5.4 임베딩 모델

### 5.4.2 유사도 계산: 코사인 유사도

```py 원고
import numpy as np

def cosine_similarity(vec1, vec2):
    dot = np.dot(vec1, vec2)
    return dot / (np.linalg.norm(vec1) * np.linalg.norm(vec2))
```

```py 보조
#! note: 다음 셀이 쓰는 embeddings_model은 5.4.3에서 만듭니다. 순서대로 실행할 수 있게 여기서 먼저 만듭니다.
from langchain_openai import OpenAIEmbeddings

embeddings_model = OpenAIEmbeddings(model="text-embedding-3-small")
```

```py 원고
query_embedding = embeddings_model.embed_query("반품 비용이 얼마예요?")
doc_embedding = embeddings_model.embed_query("환불 시 왕복 배송비는 5,000원입니다.")

similarity = cosine_similarity(
    np.array(query_embedding),
    np.array(doc_embedding),
)
print(f"유사도: {similarity:.4f}")
```

## 5.7 RAG 아키텍처

### 5.7.2 2단계 RAG

```py 원고
from langchain.chat_models import init_chat_model
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

llm = init_chat_model("gpt-4o-mini", temperature=0)

prompt = ChatPromptTemplate.from_template("""
다음 문서를 참고해서 질문에 답하세요.

문서:
{context}

질문: {question}
""")

def format_docs(docs):
    return "\n\n".join(doc.page_content for doc in docs)
```

```py 원고
rag_chain = (
    {"context": retriever | format_docs, "question": RunnablePassthrough()}
    | prompt
    | llm
    | StrOutputParser()
)
```

```py 원고
answer = rag_chain.invoke("반품 배송비 얼마야?")
print(answer)
```

### 5.7.3 에이전트형 RAG

```py 원고
from langchain_core.tools import tool
from langchain.chat_models import init_chat_model
from langchain.agents import create_agent as create_react_agent
import requests

@tool
def search_zmall_products(query: str) -> str:
    """Z몰 상품 DB에서 관련 상품을 검색합니다."""
    docs = retriever.invoke(query)
    return "\n".join(doc.page_content for doc in docs)

@tool
def fetch_webpage(url: str) -> str:
    """웹페이지에서 텍스트 정보를 가져옵니다."""
    response = requests.get(url, timeout=10)
    return response.text[:2000]
```

```py 원고
llm = init_chat_model("gpt-4o-mini", temperature=0)

agent = create_react_agent(
    model=llm,
    tools=[search_zmall_products, fetch_webpage],
)
```

```py 원고
result = agent.invoke({
    "messages": [{"role": "user", "content": "여름 린넨 원피스와 어울리는 샌들 추천해줘"}]
})
print(result["messages"][-1].content)
```

### 5.7.5 근거가 없을 때는 답하지 않는다

```py 원고
def retrieve_evidence(question: str, min_score: float = 0.75):
    """유사도 점수가 기준 이상인 문서만 근거로 남깁니다."""
    results = vector_store.similarity_search_with_score(question, k=3)
    return [doc for doc, score in results if score >= min_score]

def answer_policy_question(question: str) -> dict:
    docs = retrieve_evidence(question)
    if not docs:
        return {
            "status": "정보 부족",
            "answer": "현재 확인할 수 있는 규정이 없습니다. 담당자에게 확인 후 안내드리겠습니다.",
            "evidence": [],
        }
    answer = (prompt | llm | StrOutputParser()).invoke(
        {"context": format_docs(docs), "question": question}
    )
    return {"status": "완료", "answer": answer, "evidence": [doc.id for doc in docs]}

print(answer_policy_question("해외 배송 시 관세는 누가 내나요?"))
print(answer_policy_question("반품 배송비 얼마야?"))
```

```py 보조
#! note: 기준값(min_score)을 정하려면 질문마다 실제 점수를 먼저 봅니다.
for q in ["해외 배송 시 관세는 누가 내나요?", "반품 배송비 얼마야?"]:
    print(q)
    for doc, score in vector_store.similarity_search_with_score(q, k=3):
        print(f"  {score:.3f}  {doc.id}  {doc.page_content[:40]}")
```
