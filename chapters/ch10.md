### 10.1.3 Graph API 실습: 고객 승인 받고 재개하기

```py 원고
from typing import TypedDict, Literal, Optional

class CartState(TypedDict):
    product: str
    approved: Optional[Literal["yes", "no"]]
    result: str
```

```py 원고
from langgraph.config import get_stream_writer
from langgraph.types import interrupt

def recommend_product(state: CartState):
    writer = get_stream_writer()
    writer("추천 상품을 선택하는 중입니다...")
    return {"product": "무선 청소기", "approved": None, "result": ""}

def ask_customer_approval(state: CartState):
    writer = get_stream_writer()
    writer(f"'{state['product']}'을(를) 장바구니에 담을까요? (yes/no)")

    # 여기서 그래프 실행이 멈춥니다.
    # Command(resume=...) 입력값이 interrupt()의 반환값으로 들어옵니다.
    approval = interrupt({
        "question": "장바구니에 담을까요?",
        "options": ["yes", "no"]
    })
    return {"approved": approval}

def add_to_cart(state: CartState):
    writer = get_stream_writer()
    writer("승인됨. 장바구니에 상품을 담는 중입니다...")
    return {"result": f"{state['product']}을(를) 장바구니에 담았습니다."}

def cancel(state: CartState):
    writer = get_stream_writer()
    writer("거절됨. 작업을 종료합니다.")
    return {"result": "요청이 취소되었습니다."}

def route_by_approval(state: CartState):
    if state["approved"] == "yes":
        return "add"
    return "cancel"
```

```py 원고
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import InMemorySaver

builder = StateGraph(CartState)

builder.add_node("recommend", recommend_product)
builder.add_node("approve", ask_customer_approval)
builder.add_node("add", add_to_cart)
builder.add_node("cancel", cancel)

builder.add_edge(START, "recommend")
builder.add_edge("recommend", "approve")

builder.add_conditional_edges(
    "approve",
    route_by_approval,
    {"add": "add", "cancel": "cancel"}
)

builder.add_edge("add", END)
builder.add_edge("cancel", END)

graph = builder.compile(checkpointer=InMemorySaver())
```

```py 원고
tid = "cart-approval-demo"
config = {"configurable": {"thread_id": tid}}
inputs = {"product": "", "approved": None, "result": ""}

for mode, chunk in graph.stream(inputs, config=config, stream_mode=["custom", "updates"]):
    print("mode:", mode, "chunk:", chunk)
```

```py 원고
from langgraph.types import Command

for mode, chunk in graph.stream(
    Command(resume="yes"),
    config=config,
    stream_mode=["custom", "updates"]
):
    print("mode:", mode, "chunk:", chunk)

print("최종 상태:", graph.get_state(config).values)
```

### 10.1.4 Functional API 실습: interrupt + resume 패턴

```py 원고
from langgraph.func import entrypoint, task
from langgraph.checkpoint.memory import InMemorySaver
from langgraph.config import get_stream_writer
from langgraph.types import interrupt, Command

checkpointer = InMemorySaver()

@task
def recommend() -> str:
    return "무선 청소기"

@task
def ask_approval(product: str) -> str:
    writer = get_stream_writer()
    writer(f"'{product}'을(를) 장바구니에 담을까요? (yes/no)")
    approval = interrupt({
        "question": "장바구니에 담을까요?",
        "options": ["yes", "no"]
    })
    return approval

@task
def add_to_cart(product: str) -> str:
    return f"{product}을(를) 장바구니에 담았습니다."

@entrypoint(checkpointer=checkpointer)
def cart_flow(_: dict) -> str:
    writer = get_stream_writer()

    product = recommend().result()
    writer(f"추천 상품: {product}")

    approved = ask_approval(product).result()

    if approved == "yes":
        writer("승인됨. 장바구니 담기를 진행합니다.")
        return add_to_cart(product).result()

    writer("거절됨. 작업을 종료합니다.")
    return "요청이 취소되었습니다."
```

```py 원고
config = {"configurable": {"thread_id": "cart-func-demo"}}

for mode, chunk in cart_flow.stream({}, config=config, stream_mode=["custom", "updates"]):
    print(mode, chunk)
```

```py 원고
from langgraph.types import Command

for mode, chunk in cart_flow.stream(
    Command(resume="no"),
    config=config,
    stream_mode=["custom", "updates"]
):
    print(mode, chunk)
```

### 10.2.4 Graph API 전체 예제: 마케팅 메일 생성기

```py 원고
import uuid
import random
from typing_extensions import TypedDict, NotRequired, Literal
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import InMemorySaver

class EmailState(TypedDict):
    subject: NotRequired[str]
    body: NotRequired[str]
    last_step: NotRequired[Literal["choose_subject", "write_body"]]

subjects = ["주말 세일 안내", "신제품 출시 소식", "회원 전용 이벤트"]
endings = [
    "를 지금 확인해보세요!",
    "를 놓치지 마세요!",
    "에 대한 자세한 내용을 확인하세요.",
]

def choose_subject(state: EmailState) -> EmailState:
    return {
        "subject": random.choice(subjects),
        "last_step": "choose_subject",
    }

def write_body(state: EmailState) -> EmailState:
    return {
        "body": state["subject"] + random.choice(endings),
        "last_step": "write_body",
    }

workflow = StateGraph(EmailState)
workflow.add_node("choose_subject", choose_subject)
workflow.add_node("write_body", write_body)
workflow.add_edge(START, "choose_subject")
workflow.add_edge("choose_subject", "write_body")
workflow.add_edge("write_body", END)

checkpointer = InMemorySaver()
graph = workflow.compile(checkpointer=checkpointer)

thread_id = str(uuid.uuid4())
config = {"configurable": {"thread_id": thread_id}}

# 첫 실행: 제목 + 본문 생성
first = graph.invoke({}, config)
print("첫 실행:", first)

# 제목 선택 직후(다음 노드가 write_body인) 체크포인트를 찾아 시간여행
target = next(
    snap for snap in graph.get_state_history(config)
    if snap.next == ("write_body",)
)

# 같은 제목, 다른 본문으로 재실행
second = graph.invoke(None, target.config)
print("시간여행 결과:", second)
```

### 10.4.3 본문 생성: GPT에 맡기기

```py 원고
import uuid
from typing_extensions import TypedDict, NotRequired, Literal

from langchain.chat_models import init_chat_model
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import InMemorySaver
from langgraph.types import interrupt, Command

llm = init_chat_model("gpt-4o-mini")   # .env의 OPENAI_API_KEY를 씁니다

class EmailState(TypedDict):
    topic: str
    tone: NotRequired[str]
    body: NotRequired[str]
    subject: NotRequired[str]
    edit_choice: NotRequired[Literal["subject", "body", "both", "none"]]
    audit: NotRequired[list[str]]

def _log(state: EmailState, msg: str) -> list[str]:
    audit = state.get("audit") or []
    audit.append(msg)
    return audit

def gpt_make_body(topic: str, tone: str) -> str:
    prompt = (
        f"주제: {topic}\n"
        f"톤: {tone}\n"
        "마케팅 메일 본문을 한국어로 4~6문장으로 작성해줘. 마지막에 CTA 1문장 포함."
    )
    return llm.invoke(prompt).content.strip()

def gen_body(state: EmailState) -> EmailState:
    tone = state.get("tone") or "친근"
    body = gpt_make_body(state["topic"], tone)
    audit = _log(state, f"[gen_body] tone={tone}")
    return {"body": body, "tone": tone, "audit": audit}
```

### 10.4.4 제목 생성: 본문을 주고 제목 짓기

```py 원고
def gpt_make_subject_from_body(body: str) -> str:
    prompt = (
        "아래 마케팅 메일 본문을 읽고, 제목을 한국어로 1개만 만들어줘.\n"
        "- 20자 내외\n"
        "- 클릭을 유도하되 과장하지 말 것\n\n"
        f"본문:\n{body}"
    )
    return llm.invoke(prompt).content.strip()

def gen_subject_from_body(state: EmailState) -> EmailState:
    subject = gpt_make_subject_from_body(state["body"])
    audit = _log(state, f"[gen_subject_from_body] subject={subject}")
    return {"subject": subject, "audit": audit}
```

### 10.4.5 Interrupts: 편집 선택 받기

```py 원고
def choose_edit(state: EmailState) -> EmailState:
    choice = interrupt({
        "kind": "edit_choice",
        "question": "무엇을 바꿀까요? (subject/body/both/none)",
        "current": {"subject": state.get("subject"), "body": state.get("body")},
        "options": ["subject", "body", "both", "none"],
    })
    choice = str(choice).strip().lower()
    if choice not in ["subject", "body", "both", "none"]:
        choice = "none"
    audit = _log(state, f"[choose_edit] resume={choice}")
    return {"edit_choice": choice, "audit": audit}
```

### 10.4.6 Time Travel: 필요한 단계만 다시 실행

```py 원고
def find_checkpoint(graph, config, next_node: str):
    """다음에 실행할 노드가 next_node인 가장 최근 체크포인트의 config를 찾습니다."""
    for snap in graph.get_state_history(config):
        if snap.next == (next_node,):
            return snap.config
    raise ValueError(f"{next_node} 직전 체크포인트가 없습니다")

def rerun_by_choice(graph, config, choice: str):
    if choice == "subject":
        # 본문까지 만든 시점(다음 노드가 gen_subject_from_body)에서 재실행
        return graph.invoke(None, find_checkpoint(graph, config, "gen_subject_from_body"))

    if choice == "both":
        # 맨 처음 시점(다음 노드가 gen_body)에서 재실행
        return graph.invoke(None, find_checkpoint(graph, config, "gen_body"))

    if choice == "body":
        # 제목까지 만든 시점에서 본문만 바꾼 분기를 만들고 재실행
        target = find_checkpoint(graph, config, "choose_edit")
        values = graph.get_state(target).values
        new_body = gpt_make_body(values["topic"], values.get("tone") or "친근")
        forked = graph.update_state(target, {"body": new_body}, as_node="gen_subject_from_body")
        return graph.invoke(None, forked)

    return None
```

### 10.4.7 그래프 조립과 실행

```py 원고
# 10.4.3~10.4.6의 조각에 이어서 실행합니다.
builder = StateGraph(EmailState)
builder.add_node("gen_body", gen_body)
builder.add_node("gen_subject_from_body", gen_subject_from_body)
builder.add_node("choose_edit", choose_edit)

builder.add_edge(START, "gen_body")
builder.add_edge("gen_body", "gen_subject_from_body")
builder.add_edge("gen_subject_from_body", "choose_edit")
builder.add_edge("choose_edit", END)

graph = builder.compile(checkpointer=InMemorySaver())

thread_id = str(uuid.uuid4())
config = {"configurable": {"thread_id": thread_id}}

# 1) 첫 실행: 본문/제목 생성 후 interrupt에서 멈춤
out = graph.invoke({"topic": "주말 세일", "audit": []}, config=config)
print("INTERRUPT:", out.get("__interrupt__"))

# 2) 사용자가 "subject"를 선택 (값을 바꿔 테스트해 보세요)
out2 = graph.invoke(Command(resume="subject"), config=config)
choice = out2.get("edit_choice", "none")
print("CHOICE:", choice)

# 3) Time Travel로 재실행(새 분기 생성). 분기한 실행도 choose_edit에서 다시 멈춥니다.
forked = rerun_by_choice(graph, config, choice)
if forked:
    print("FORKED SUBJECT:", forked.get("subject"))
    print("FORKED BODY:", forked.get("body"))

# 4) 현재 상태 확인
print("CURRENT STATE:", graph.get_state(config).values)
```
