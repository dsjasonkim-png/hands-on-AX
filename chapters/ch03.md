## 3.1 LLM 모델과의 상호작용

### 3.1.1 모델 초기화

```py 원고
from langchain.chat_models import init_chat_model
from dotenv import load_dotenv

# .env 파일의 OPENAI_API_KEY를 환경 변수로 불러옵니다 (2장 참고)
load_dotenv()

# GPT-4o 모델을 초기화합니다
model = init_chat_model("gpt-4o")
```

### 3.1.2 하이퍼파라미터로 모델 동작 조정하기

```py 원고
from langchain.chat_models import init_chat_model

# 상황 A: 마케팅팀의 기획전 문구 - 다양한 표현이 필요합니다
copy_model = init_chat_model(
    "gpt-4o",
    temperature=0.9, # 다양하고 자유로운 표현을 허용합니다
    max_tokens=150, # 문구는 짧게 쓰고 멈춥니다
    timeout=10
)

# 상황 B: 고객지원팀의 문의 분류 - 같은 입력에 같은 답이 나와야 합니다
triage_model = init_chat_model(
    "gpt-4o-mini",
    temperature=0.0, # 가장 확률 높은 답변을 고릅니다
    max_tokens=300, # 분류 결과와 이유만 받습니다
    timeout=30
)
```

```py 보조
#! note: 두 모델에 같은 요청을 보내 temperature와 max_tokens의 차이를 봅니다.
ask = "Z몰 여름 기획전 한 줄 문구를 써줘"
for name, m in [("copy_model", copy_model), ("triage_model", triage_model)]:
    print(name, "→", m.invoke(ask).content)
```

### 3.1.3 모델 호출 방식: invoke, stream, batch

```py 원고
chat_log = """
고객: 지난주에 주문한 운동화가 아직 안 왔어요.
상담사: 주문번호를 알려주시겠어요?
고객: ZM-1029요. 다음 주 월요일까지 꼭 받아야 해요.
"""
response = model.invoke(f"다음 상담 내용을 담당자 이관용 메모로 3줄 요약해줘:\n{chat_log}")
print(response.content)
```

```py 원고
for chunk in model.stream("배송이 지연된 고객에게 보낼 사과 안내문을 써줘"):
    print(chunk.content, end="", flush=True)
    # flush=True는 버퍼에 쌓지 않고 즉시 화면에 출력합니다
```

```py 원고
reviews = [
    "배송이 너무 늦어서 화나요.",
    "상품 품질이 기대 이상입니다!",
    "그저 그렇네요. 보통입니다."
]

prompts = [f"다음 리뷰의 감정을 긍정·부정·중립 중 하나로 고르고 이유를 한 문장으로 써줘: {r}" for r in reviews]

results = model.batch(
    prompts,
    config={"max_concurrency": 5}, # 동시에 보내는 요청 수를 제한합니다
)

for i, result in enumerate(results):
    print(f"리뷰 {i+1} 분석 결과: {result.content}")
```

## 3.2 메시지 시스템 구조

### 3.2.2 메시지의 4가지 종류

```py 원고
from langchain_core.messages import SystemMessage

sys_msg = SystemMessage(content="""
당신은 Z몰 고객지원 담당입니다.
주문 조회 결과에 없는 내용은 답하지 않고,
확인이 필요하면 고객에게 되물어 주세요.
존댓말을 사용해 주세요.
""")
```

```py 원고
from langchain_core.messages import HumanMessage

human_msg = HumanMessage(
    content="ZM-1029 주문 배송 상태 알려주세요.",
    name="user_1029", # 누가 보냈는지 식별합니다
    id="msg_001" # 나중에 추적하기 위한 ID입니다
)
```

```py 보조
#! note: 본문의 AIMessage 예시는 도구를 연결한 모델의 응답(ai_response)을 출력합니다. 4장에서 만들 주문 조회 도구를 미리 간단히 붙여 그 응답을 만듭니다.
from langchain_core.tools import tool

@tool
def lookup_order(order_id: str) -> str:
    """주문번호로 배송 상태를 조회합니다."""
    return f"주문 {order_id}: 배송 중, 11/09 도착 예정"

ai_response = model.bind_tools([lookup_order]).invoke([sys_msg, human_msg])
```

```py 원고
# 모델 응답 예시
print(ai_response.content)
# 출력: ""   (도구를 호출할 때는 본문이 비어 있는 경우가 많습니다)

print(ai_response.tool_calls)
# 출력: [{'name': 'lookup_order', 'args': {'order_id': 'ZM-1029'}, 'id': 'call_abc123', 'type': 'tool_call'}]

print(ai_response.usage_metadata)
# 출력: {'input_tokens': 50, 'output_tokens': 20, 'total_tokens': 70}
```

```py 원고
from langchain_core.messages import ToolMessage

tool_msg = ToolMessage(
    content="주문 ZM-1029: 배송 중, 11/09 도착 예정",
    tool_call_id="call_abc123", # AIMessage의 tool_calls에 있던 id
)
```

```py 보조
#! note: 위 셀의 call_abc123 대신 실제 요청 ID를 넣어, 도구 결과까지 받은 모델이 최종 답변을 만드는 것을 봅니다.
tool_msg = ToolMessage(content="주문 ZM-1029: 배송 중, 11/09 도착 예정",
                       tool_call_id=ai_response.tool_calls[0]["id"])
final = model.invoke([sys_msg, human_msg, ai_response, tool_msg])
print(final.content)
```

### 3.2.4 대화 기록 관리

```py 원고
from langchain_core.messages import (
    SystemMessage, HumanMessage, AIMessage, trim_messages
)

history = [
    SystemMessage("당신은 Z몰 고객지원 담당입니다."),
    HumanMessage("안녕하세요"),
    AIMessage("무엇을 도와드릴까요?"),
    HumanMessage("제 주문 언제 와요?"),
    AIMessage("주문번호를 알려주시겠어요?"),
    HumanMessage("ZM-1029요"),
]

trimmed = trim_messages(
    history,
    strategy="last", # 최근 메시지부터 남깁니다
    token_counter=len, # 예제에서는 메시지 개수로 셉니다
    max_tokens=4, # 최대 4개만 남깁니다
    include_system=True, # 업무 지침은 항상 유지합니다
    start_on="human", # 남은 기록이 고객 메시지로 시작하게 합니다
)

for m in trimmed:
    print(type(m).__name__, ":", m.content)
```

### 3.2.5 멀티모달: 이미지와 텍스트를 함께 보내기

```py 읽기전용
#! note: 본문의 이미지 주소(example.com)는 예시라 실제 이미지가 없습니다. 아래 셀이 17장 실습용 영수증 이미지를 Base64로 보내는 버전입니다.
from langchain_core.messages import HumanMessage

multimodal_msg = HumanMessage(
    content=[
        # 텍스트 블록
        {"type": "text", "text": "이 영수증의 상호, 결제 금액, 항목을 읽어줘."},
        # 이미지 블록
        {"type": "image_url", "image_url": {"url": "https://example.com/receipt_sample.jpg"}}
    ]
)

response = model.invoke([multimodal_msg])
print(response.content)
```

```py 보조
import base64
from pathlib import Path
from langchain_core.messages import HumanMessage

image = Path("../../axcore_proj/samples/receipt/R002_meeting_meal.png")
data_url = "data:image/png;base64," + base64.b64encode(image.read_bytes()).decode()

multimodal_msg = HumanMessage(
    content=[
        {"type": "text", "text": "이 영수증의 상호, 결제 금액, 항목을 읽어줘."},
        {"type": "image_url", "image_url": {"url": data_url}},
    ]
)
response = model.invoke([multimodal_msg])
print(response.content)
```

### 3.2.6 표준 콘텐츠 블록

```py 원고
# 4장에서 도구를 연결한 모델(model_with_tools)이라면 tool_call 블록이 함께 나옵니다
response = model.invoke("ZM-1029 주문 배송 상태 알려줘")

for block in response.content_blocks:
    if block["type"] == "text":
        print("답변:", block["text"])
    elif block["type"] == "tool_call":
        print("도구 요청:", block["name"], block["args"])
```
