## 11.1 미들웨어

```py 보조
# 4장에서 만든 주문 조회 도구와 4.1.7의 환불 요청서 도구를 실행용으로 간단히 다시 만듭니다.
# 본문처럼 첫 번째 주문 조회는 일부러 시간 초과로 실패하게 해 두었습니다.
from langchain_core.tools import tool

order_api_calls = {"n": 0}

@tool
def lookup_order(order_id: str) -> str:
    """Z몰 주문 DB에서 주문번호로 주문 상태를 조회합니다."""
    order_api_calls["n"] += 1
    if order_api_calls["n"] == 1:
        raise TimeoutError("주문 조회 API 시간 초과")
    return f"주문 {order_id}: 배송 중, 11/09 도착 예정"

@tool
def request_refund(order_id: str, reason: str) -> str:
    """환불 요청서를 만듭니다. 실제 환불은 담당자 승인 뒤에 진행됩니다."""
    return f"{order_id} 환불 요청서 접수(사유: {reason}). 담당자 승인 대기"
```

### 11.1.3 실행 기록과 도구 재시도

```py 원고
from langchain.agents.middleware import AgentMiddleware

class SessionLogMiddleware(AgentMiddleware):
    """문의 한 건의 시작과 끝을 기록합니다."""

    def before_agent(self, state, runtime):
        print("[시작]", state["messages"][-1].content)

    def after_agent(self, state, runtime):
        n = sum(len(getattr(m, "tool_calls", None) or []) for m in state["messages"])
        print(f"[끝] 도구 호출 {n}회")
```

```py 원고
class OrderApiRetryMiddleware(AgentMiddleware):
    """주문 조회 API가 시간 초과로 실패하면 두 번까지 다시 호출합니다."""

    def wrap_tool_call(self, request, handler):
        for attempt in range(3):
            try:
                return handler(request)            # 실제 도구 실행
            except TimeoutError:
                print(f"{request.tool_call['name']} 재시도 {attempt + 1}")
                if attempt == 2:
                    raise
```

```py 원고
from dataclasses import dataclass
from langchain.agents import create_agent as create_react_agent
from langchain.agents.middleware import dynamic_prompt, ModelRequest
from langgraph.store.memory import InMemoryStore

@dataclass
class ZMallContext:
    user_id: str
    user_role: str = "customer"   # "customer" 또는 "admin"
    region: str = "KR"            # 접속 지역 (11.3.2)
    mode: str = "chat"            # "chat" 또는 "ticket" (11.3.5)

@dynamic_prompt
def z_mall_smart_prompt(request: ModelRequest) -> str:
    base = "당신은 Z몰 고객지원 상담원입니다. 조회 결과에 없는 내용은 답하지 않습니다."

    # [상태] 대화 길이에 따른 제어
    if len(request.messages) > 10:
        base += "\n대화가 길어지고 있습니다. 핵심만 간결하게 답변하세요."

    # [저장소] VIP 여부에 따른 제어
    user_prefs = request.runtime.store.get(
        ("preferences",), request.runtime.context.user_id
    )
    if user_prefs and user_prefs.value.get("is_vip"):
        base += "\n이 고객은 VIP입니다. 전담 상담 연결이 가능함을 함께 안내하세요."

    # [런타임 컨텍스트] 사용자 구분에 따른 제어
    if request.runtime.context.user_role == "admin":
        base += "\n관리자 테스트 중입니다. 실제 환불·주문 처리를 안내하지 마세요."

    return base

agent = create_react_agent(
    model="gpt-4o-mini",
    tools=[lookup_order],
    middleware=[z_mall_smart_prompt],
    context_schema=ZMallContext,
    store=InMemoryStore(),
)
```

### 11.3.2 메시지 일회성 주입

```py 원고
from langchain.agents.middleware import wrap_model_call, ModelRequest, ModelResponse
from typing import Callable

@wrap_model_call
def inject_zmall_context(
    request: ModelRequest,
    handler: Callable[[ModelRequest], ModelResponse],
) -> ModelResponse:
    uploaded_files = request.state.get("uploaded_files", [])
    region = request.runtime.context.region

    extra = []
    if uploaded_files:
        file_list = ", ".join(f["name"] for f in uploaded_files)
        extra.append(f"고객 첨부 파일: {file_list}")
    if region == "KR":
        extra.append(
            "반품 규정: 전자상거래법에 따라 단순 변심은 수령 후 7일 이내만 가능합니다."
        )

    if extra:
        injected = {
            "role": "user",
            "content": "[시스템 알림]\n" + "\n".join(extra),
        }
        request = request.override(messages=[*request.messages, injected])

    return handler(request)
```

### 11.3.3 도구 권한 기반 필터링

```py 원고
@wrap_model_call
def filter_tools_by_auth(
    request: ModelRequest,
    handler: Callable[[ModelRequest], ModelResponse],
) -> ModelResponse:
    is_authenticated = request.state.get("authenticated", False)
    user_role = request.runtime.context.user_role

    allowed = []
    for tool in request.tools:
        if tool.name == "request_refund":     # 4.1.7의 변경 도구
            if user_role == "admin" or is_authenticated:
                allowed.append(tool)
        else:
            allowed.append(tool)

    return handler(request.override(tools=allowed))
```

### 11.3.4 모델 동적 선택

```py 원고
from langchain.chat_models import init_chat_model

heavy_model = init_chat_model("gpt-4o")
light_model = init_chat_model("gpt-4o-mini")

@wrap_model_call
def optimize_model_selection(
    request: ModelRequest,
    handler: Callable[[ModelRequest], ModelResponse],
) -> ModelResponse:
    user_prefs = request.runtime.store.get(("preferences",), request.runtime.context.user_id)
    is_premium = user_prefs.value.get("is_premium", False) if user_prefs else False

    # 프리미엄 고객이거나 대화가 길어 복잡한 문맥 파악이 필요할 때
    if is_premium or len(request.messages) > 15:
        model = heavy_model
    else:
        model = light_model

    return handler(request.override(model=model))
```

```py 원고
from typing import Literal
from pydantic import BaseModel, Field

class ZMallSupportTicket(BaseModel):
    category: Literal["배송지연", "제품불량", "결제오류", "기타"] = Field(description="불만 유형")

def enforce_ticket_format(
    request: ModelRequest,
    handler: Callable[[ModelRequest], ModelResponse],

        request = request.override(response_format=ZMallSupportTicket)
    return handler(request)
```

```py 읽기전용
#! note: 배송사 서버 주소(delivery-api.zmall.com)와 서버 파일 경로가 예시라서 그대로는 실행되지 않습니다. 아래 셀이 실행용으로 바꾼 버전입니다.
import asyncio
```

```py 원고
import asyncio

async def retry_interceptor(request: MCPToolCallRequest, handler, max_retries: int = 3):
    last_error = None
    for attempt in range(max_retries):
        try:
            return await handler(request)
        except Exception as e:
            last_error = e
            if attempt < max_retries - 1:
                wait_time = 1.0 * (2 ** attempt)  # 1초, 2초, 4초...
                print(f"{request.name} 실패. {wait_time}초 후 재시도합니다.")
                await asyncio.sleep(wait_time)
    raise last_error
```

```py 보조
from langchain_mcp_adapters.callbacks import Callbacks, CallbackContext

async def on_progress(
    progress: float,
    total: float | None,
    message: str | None,
    ctx: CallbackContext,
):
    percent = (progress / total * 100) if total else progress
    print(f"[{ctx.server_name}] {percent:.1f}% - {message}")

client = MultiServerMCPClient(
    {"zmall_internal": ZMALL_SERVER},
    callbacks=Callbacks(on_progress=on_progress),
)
tools = {t.name: t for t in await client.get_tools()}
```

```py 읽기전용
# 서버 측: 도구 실행 중 추가 정보 요청
from pydantic import BaseModel
from mcp.server.fastmcp import Context, FastMCP

server = FastMCP("Profile")

class RecipientDetails(BaseModel):
    email: str
    age: int

@server.tool()
async def create_gift_profile(recipient_name: str, ctx: Context) -> str:
    """선물 수신자 프로필을 생성합니다."""
    result = await ctx.elicit(
        message=f"{recipient_name}님의 프로필을 만들려면 추가 정보가 필요합니다:",
        schema=RecipientDetails,
    )
    if result.action == "accept" and result.data:
        return f"프로필 생성 완료: {result.data.email}, {result.data.age}세"
    if result.action == "decline":
        return "사용자가 정보 제공을 거부했습니다. 기본 프로필만 생성합니다."
    return "프로필 생성이 취소되었습니다."
```
