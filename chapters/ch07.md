### 7.2.2 Graph App의 생성과 실행

```py 원고
from langgraph.graph import StateGraph, START, END
from typing_extensions import TypedDict

class State(TypedDict):
    score: float
    result: str

def node_a(state: State):
    return {"result": f"정규화된 점수: {state['score']}\n조건부 분기: A 노드입니다."}

def node_b(state: State):
    return {"result": f"정규화된 점수: {state['score']}\n조건부 분기: B 노드입니다."}

def normalize_score(state: State):
    return {"score": state["score"] / 100}
```

```py 원고
graph_builder = StateGraph(State)

graph_builder.add_node("normalize", normalize_score)
graph_builder.add_node("node_a", node_a)
graph_builder.add_node("node_b", node_b)

graph_builder.add_edge(START, "normalize")
graph_builder.add_conditional_edges(
    "normalize",
    lambda state: "A" if state["score"] > 0.5 else "B",
    {"A": "node_a", "B": "node_b"},
)
graph_builder.add_edge("node_a", END)
graph_builder.add_edge("node_b", END)

app = graph_builder.compile()
```

```py 원고
output = app.invoke({"score": 75, "result": ""})
print(output["result"])
```

```py 원고
print(app.invoke({"score": 40, "result": ""})["result"])
```

```py 원고
for event in app.stream({"score": 75, "result": ""}):
    print(event)
```

```py 원고
from typing import TypedDict
from dotenv import load_dotenv
from langchain.chat_models import init_chat_model
from langchain_core.messages import SystemMessage, HumanMessage
from langgraph.graph import StateGraph, START, END

load_dotenv()
llm = init_chat_model("gpt-4o-mini")

class State(TypedDict, total=False):
    score: float
    context: str
    prompt: str
    answer: str
```

```py 원고
def normalize_score(state: State):
    return {"score": state["score"] / 100}

def node_a(state: State):
    return {"context": f"정규화된 점수: {state['score']}\n분기: A 노드입니다."}

def node_b(state: State):
    return {"context": f"정규화된 점수: {state['score']}\n분기: B 노드입니다."}

def build_prompt(state: State):
    prompt = f"""[컨텍스트]
{state["context"]}

[요청]
LangGraph의 실행 결과입니다. 이 결과를 한 문단으로 독자들에게 설명해주세요."""
    return {"prompt": prompt, "answer": ""}
```

```py 원고
def generate_with_llm_stream(state: State):
    messages = [
        SystemMessage(content="데이터/AI 초보자에게 친절하게 설명하는 조교입니다."),
        HumanMessage(content=state["prompt"]),
    ]
    answer = state.get("answer", "")
    for chunk in llm.stream(messages):
        token = chunk.content or ""
        if not token:
            continue
        answer += token
        yield {"answer": answer}
```

```py 원고
graph_builder = StateGraph(State)

graph_builder.add_node("normalize", normalize_score)
graph_builder.add_node("node_a", node_a)
graph_builder.add_node("node_b", node_b)
graph_builder.add_node("build_prompt", build_prompt)
graph_builder.add_node("generate", generate_with_llm_stream)

graph_builder.add_edge(START, "normalize")
graph_builder.add_conditional_edges(
    "normalize",
    lambda state: "A" if state["score"] > 0.5 else "B",
    {"A": "node_a", "B": "node_b"},
)
graph_builder.add_edge("node_a", "build_prompt")
graph_builder.add_edge("node_b", "build_prompt")
graph_builder.add_edge("build_prompt", "generate")
graph_builder.add_edge("generate", END)

app = graph_builder.compile()
```

```py 원고
for chunk, metadata in app.stream({"score": 75}, stream_mode="messages"):
    if chunk.content:
        print(chunk.content, end="", flush=True)
```

### 7.3.2 @task와 @entrypoint

```py 원고
from langgraph.func import task
from langchain.chat_models import init_chat_model

llm = init_chat_model("gpt-4o-mini")

def summarize_text(text: str) -> str:
    """텍스트를 한 문장으로 요약합니다."""
```

### 7.3.4 실전 예제: 요약 + 말투 변환 워크플로

```py 원고
from langgraph.func import entrypoint, task
from langgraph.checkpoint.memory import InMemorySaver
from langchain.chat_models import init_chat_model

llm = init_chat_model("gpt-4o-mini")

@task
def summarize_text(text: str) -> str:
    return llm.invoke(f"요약해줘:\n{text}").content

@task
def to_friendly_tone(text: str) -> str:
    return llm.invoke(f"친근한 말투로 다시 써줘:\n{text}").content
```

```py 원고
checkpointer = InMemorySaver()

@entrypoint(checkpointer=checkpointer)
def workflow(inputs: dict):
    summary = summarize_text(inputs["text"]).result()
    friendly = to_friendly_tone(summary).result()
    return friendly
```

```py 원고
config = {"configurable": {"thread_id": "demo"}}

result = workflow.invoke(

LangGraph is a low-level orchestration framework for building, managing,

"""},
    config=config,
)
```

### 7.3.5 Task 설계 원칙: 직렬화, 결정성, 멱등성

```py 원고
import uuid

# 권장하지 않음: 엔트리포인트에서 직접 만든 값은 재개할 때마다 달라짐
@entrypoint(checkpointer=checkpointer)
def workflow(inputs: dict):
    request_id = str(uuid.uuid4())
    ...

# 권장: task 안에서 만들면 결과가 저장되어 재개해도 같은 값을 씀

    request_id = new_request_id().result()
    ...
```

### 7.4.2 Graph API를 사용해야 하는 상황

```py 원고
from langgraph.graph import StateGraph, START, END
from typing import TypedDict, Literal

class State(TypedDict):
    user_input: str
    intent: str
    result: str

def classify_intent(state: State):
    text = state["user_input"]
    if "추천" in text:
        return {"intent": "recommend"}
    elif "환불" in text:
        return {"intent": "refund"}
    else:
        return {"intent": "faq"}

def recommend_node(state: State):
    return {"result": "추천 결과입니다."}

def refund_node(state: State):
    return {"result": "환불 정책 안내입니다."}

def faq_node(state: State):
    return {"result": "FAQ 답변입니다."}

def route(state: State) -> Literal["recommend", "refund", "faq"]:
    return state["intent"]
```

```py 원고
graph = StateGraph(State)

graph.add_node("classify_intent", classify_intent)
graph.add_node("recommend", recommend_node)
graph.add_node("refund", refund_node)
graph.add_node("faq", faq_node)

graph.add_edge(START, "classify_intent")
graph.add_conditional_edges("classify_intent", route)
graph.add_edge("recommend", END)
graph.add_edge("refund", END)
graph.add_edge("faq", END)

app = graph.compile()
result = app.invoke({"user_input": "청소기 추천해줘", "intent": "", "result": ""})
print(result["result"])
```

### 7.4.3 Functional API를 사용해야 하는 상황

```py 원고
from langgraph.func import entrypoint, task

@task
def load_docs(path: str) -> list:
    return ["문서1", "문서2"]

@task
def chunk_docs(docs: list) -> list:
    return [d[:100] for d in docs]

@task
def embed(chunks: list) -> list:
    return [[0.1, 0.2]] * len(chunks)

@entrypoint()
def build_index(inputs: dict):
    docs = load_docs(inputs["path"]).result()
    chunks = chunk_docs(docs).result()
    vectors = embed(chunks).result()
    return {"num_docs": len(docs), "num_chunks": len(chunks)}
```
