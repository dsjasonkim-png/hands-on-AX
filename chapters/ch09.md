### 9.1.4 Graph API 예제: 과일 바구니

```py 원고
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import InMemorySaver
from typing import TypedDict

# 상태 정의: 바구니 안 과일 목록을 저장합니다.
class BasketState(TypedDict):
    fruits: list[str]

# 노드 1: 바구니에 바나나 추가
def add_banana(state: BasketState):
    return {"fruits": state["fruits"] + ["banana"]}

# 노드 2: 바구니에 사과 추가
def add_apple(state: BasketState):
    return {"fruits": state["fruits"] + ["apple"]}

# 그래프 구성
builder = StateGraph(BasketState)
builder.add_node("add_banana", add_banana)
builder.add_node("add_apple", add_apple)

builder.add_edge(START, "add_banana")
builder.add_edge("add_banana", "add_apple")
builder.add_edge("add_apple", END)

# 체크포인터를 지정해서 컴파일합니다.
graph = builder.compile(checkpointer=InMemorySaver())

# thread_id를 설정하고 실행합니다.
config = {"configurable": {"thread_id": "fruit-demo"}}
graph.invoke({"fruits": []}, config)

# 현재 상태 확인
print(graph.get_state(config).values)

# 상태 히스토리 확인
history = [snap.values for snap in graph.get_state_history(config)]
print(history)
```

### 9.1.5 Functional API 예제: 과일 바구니

```py 원고
from langgraph.func import entrypoint, task
from langgraph.checkpoint.memory import InMemorySaver

@task
def add_banana(fruits: list[str]) -> list[str]:
    return fruits + ["banana"]

@task
def add_apple(fruits: list[str]) -> list[str]:
    return fruits + ["apple"]

checkpointer = InMemorySaver()

@entrypoint(checkpointer=checkpointer)
def basket_flow(fruits: list[str]) -> list[str]:
    # @task는 future-like 객체를 반환하므로 .result()로 값을 꺼냅니다.
    with_banana = add_banana(fruits).result()
    with_apple = add_apple(with_banana).result()
    return with_apple

config = {"configurable": {"thread_id": "fruit-func"}}
result = basket_flow.invoke([], config=config)
print(result)
```

### 9.2.5 Graph API 실습: 실패 후 재개

```py 원고
import time
from typing import TypedDict

from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import InMemorySaver

class DownloadState(TypedDict):
    progress: int
    failed_once: bool

def slow_part(state: DownloadState) -> DownloadState:
    time.sleep(1)
    return {"progress": state["progress"] + 1}

def maybe_fail(state: DownloadState) -> DownloadState:
    if not state["failed_once"]:
        raise RuntimeError("일시적 오류 발생")
    return {"progress": state["progress"]}

builder = StateGraph(DownloadState)
builder.add_node("slow_part", slow_part)
builder.add_node("maybe_fail", maybe_fail)

builder.add_edge(START, "slow_part")
builder.add_edge("slow_part", "maybe_fail")
builder.add_edge("maybe_fail", END)

graph = builder.compile(checkpointer=InMemorySaver())

config = {"configurable": {"thread_id": "download-graph"}}

# 1차 실행: maybe_fail에서 에러 발생
try:
    graph.invoke(
        {"progress": 0, "failed_once": False},
        config,
        durability="async"
    )
except Exception as e:
    print("첫 실행에서 에러 발생:", e)

print("저장된 상태:", graph.get_state(config).values)

# 2차 실행: 원인을 고쳤다고 보고 상태만 바꾼 뒤, 입력 없이(None) 재개
graph.update_state(config, {"failed_once": True})
result = graph.invoke(None, config, durability="async")
print("두 번째 실행 결과:", result)
```

### 9.2.6 Functional API 실습: task 기반 재개

```py 원고
import time
from typing import TypedDict

from langgraph.func import entrypoint, task
from langgraph.checkpoint.memory import InMemorySaver

checkpointer = InMemorySaver()

class DownloadInput(TypedDict):
    steps: int

@task
def download_part(progress: int) -> int:
    """다운로드를 일부 진행하고(1초 지연), 진행률을 1 증가시킵니다."""
    time.sleep(1)
    return progress + 1

@entrypoint(checkpointer=checkpointer)
def download_workflow(inputs: DownloadInput, *, previous: int | None = None) -> int:
    """직전 실행의 진행률(previous)에 이어 steps만큼 다운로드를 진행합니다."""
    progress = previous or 0
    for _ in range(inputs["steps"]):
        progress = download_part(progress).result()
    return progress
```

```py 원고
config = {"configurable": {"thread_id": "download-demo"}}

# 1차 실행: 1단계만 진행
progress = download_workflow.invoke(
    {"steps": 1},
    config=config,
    durability="async",
)
print("1차 실행 진행 상황:", progress)

# 2차 실행: 동일 thread_id로 2단계 추가 진행
final_progress = download_workflow.invoke(
    {"steps": 2},
    config=config,
    durability="async",
)
print("2차 실행 최종 진행 상황:", final_progress)
```
