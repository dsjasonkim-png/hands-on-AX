## 13.2 서브 에이전트

### 기본 구현: Tool per Agent

```py 원고
from langchain.agents import create_agent as create_react_agent
from langchain_core.tools import tool

# 배송만 담당하는 서브 에이전트 (도구와 지침이 짧다)
delivery_subagent = create_react_agent(
    model="gpt-4o-mini",
    tools=[lookup_order],                      # 4장의 주문 조회 도구
    system_prompt="당신은 Z몰 배송 조회 담당입니다. 조회 결과만 짧게 보고합니다.",
)

# 서브 에이전트를 도구(Tool)로 포장하기
@tool("check_delivery", description="고객의 주문 배송 상태를 조회할 때 사용합니다.")
def call_delivery_agent(query: str) -> str:
    """메인 에이전트가 이 함수를 호출하면 서브 에이전트가 실행됩니다."""
    result = delivery_subagent.invoke({
        "messages": [{"role": "user", "content": query}]
    })
    return result["messages"][-1].content

# 메인 에이전트는 'check_delivery' 도구를 통해 서브 에이전트를 활용합니다.
main_agent = create_react_agent(model="gpt-4o-mini", tools=[call_delivery_agent])
```

## 13.3 핸드오프

### 13.3.2 단일 에이전트와 미들웨어

```py 원고
from typing import NotRequired
from langchain_core.tools import tool
from langchain_core.messages import ToolMessage
from langchain.tools import ToolRuntime
from langchain.agents import AgentState
from langgraph.types import Command

class ReturnState(AgentState):
    current_step: NotRequired[str]        # verification → triage → refund

# 주문 확인 도구 (성공 시 '심사' 단계로 이동)
@tool
def verify_order(order_id: str, runtime: ToolRuntime) -> Command | str:
    """주문 번호를 확인하고 유효하면 심사 단계로 넘깁니다."""
    is_valid = True  # 실제로는 DB 조회와 본인 주문 여부 확인

    if not is_valid:
        return "주문 번호가 올바르지 않습니다."
    return Command(
        update={
            "messages": [
                ToolMessage(content=f"주문 {order_id} 확인 완료.", tool_call_id=runtime.tool_call_id)
            ],
            "current_step": "triage",     # 상태를 '심사' 단계로 변경
        }
    )

# 반품 심사 도구 (성공 시 '환불' 단계로 이동)
@tool
def assess_reason(reason: str, runtime: ToolRuntime) -> Command:
    """반품 사유를 기록하고 환불 단계로 넘깁니다."""
    return Command(
        update={
            "messages": [
                ToolMessage(content=f"사유 기록됨: {reason}", tool_call_id=runtime.tool_call_id)
            ],
            "current_step": "refund",     # 상태를 '환불' 단계로 변경
        }
    )
```

```py 원고
from langchain.agents import create_agent as create_react_agent
from langchain.agents.middleware import wrap_model_call

STEP_TOOLS = {
    "verification": ["verify_order"],
    "triage": ["assess_reason"],
    "refund": ["request_refund"],         # 4.1.7의 환불 요청서 도구
}

@wrap_model_call
def tools_by_step(request, handler):
    step = request.state.get("current_step", "verification")
    allowed = [t for t in request.tools if t.name in STEP_TOOLS[step]]
    return handler(request.override(tools=allowed))

agent = create_react_agent(
    model="gpt-4o-mini",
    tools=[verify_order, assess_reason, request_refund],
    middleware=[tools_by_step],
    state_schema=ReturnState,
)
```

```py 보조

for m in result["messages"]:
    print(f"[{m.type}]", m.content or [(c["name"], c["args"]) for c in m.tool_calls])
```

### 13.3.3 다중 에이전트 그래프

```py 읽기전용
#! note: 판매·기술지원 에이전트를 노드로 둔 상위 그래프 안에서만 동작하는 도구라 이 셀만으로는 실행하지 않습니다.
from langchain_core.tools import tool
from langchain_core.messages import ToolMessage
from langchain.tools import ToolRuntime
from langgraph.types import Command

@tool
def transfer_to_support(runtime: ToolRuntime) -> Command:
    """기술 지원 에이전트에게 대화를 넘깁니다."""
    last_ai_message = runtime.state["messages"][-1]

    transfer_msg = ToolMessage(
        content="기술 지원 팀으로 연결해 드리겠습니다.",
        tool_call_id=runtime.tool_call_id
    )

    return Command(
        goto="support_agent",  # 그래프의 'support_agent' 노드로 이동
        update={
            "active_agent": "support_agent",
            "messages": [last_ai_message, transfer_msg]
        },
        graph=Command.PARENT
    )
```

```py 원고
from langchain_core.tools import tool

# 스킬 저장소 (실제로는 파일이나 DB에 저장됨)
SKILL_STORE = {
    "refund_policy": """
    [스킬: 반품 및 환불 전문가]
    당신은 지금부터 반품 규정 전문가입니다.

    - 의류는 7일 이내 반품 가능합니다.

@tool
def load_skill(skill_name: str) -> str:

        return "시스템 알림: 해당 스킬을 찾을 수 없습니다."
```

## 13.5 라우터

### 13.5.2 단일 경로 라우팅

```py 원고
from langgraph.types import Command

def classify_query(query: str) -> str:
    """질문을 분석하고 담당할 전문가를 결정합니다."""
    if "노트북" in query or "맥북" in query:
        return "electronics_agent"
    elif "티셔츠" in query:
        return "clothing_agent"
    else:
        return "general_agent"

def route_query(state):
    active_agent = classify_query(state["query"])
    print(f"안내 데스크: 고객님을 {active_agent}에게 안내합니다.")
    return Command(goto=active_agent)
```

```py 보조
# 라우터 노드와 전문가 노드 셋으로 그래프를 만들어 실행합니다.
import operator
from typing import Annotated, TypedDict
from langgraph.graph import StateGraph, START, END
from langchain.chat_models import init_chat_model

llm = init_chat_model("gpt-4o-mini")

class RouterState(TypedDict, total=False):
    query: str
    results: Annotated[list[str], operator.add]
    final_answer: str

def make_expert(role):
    def expert(state):
        answer = llm.invoke(f"당신은 Z몰 {role} 전문가입니다. 두 문장 이내로 답하세요.\n질문: {state['query']}").content
        return {"results": [f"[{role}] {answer}"]}
    return expert

builder = StateGraph(RouterState)
builder.add_node("router", route_query)
for name, role in [("electronics_agent", "전자제품"), ("clothing_agent", "의류"), ("general_agent", "일반 상담")]:
    builder.add_node(name, make_expert(role))
    builder.add_edge(name, END)
builder.add_edge(START, "router")
router_graph = builder.compile()

print(router_graph.invoke({"query": "맥북 프로 가격 알려줘"})["results"])
```

### 13.5.3 다중 경로 병렬 라우팅

```py 원고
from langgraph.types import Send

def classify_query_parallel(query: str):
    """질문을 분석하여 필요한 모든 전문가 목록을 반환합니다."""
    return [
        {"agent": "clothing_agent", "sub_query": "여름 휴가용 원피스 추천"},
        {"agent": "electronics_agent", "sub_query": "방수 카메라 추천"}
    ]

def route_query_parallel(state):
    classifications = classify_query_parallel(state["query"])
    print(f"안내 데스크: {len(classifications)}명의 전문가를 동시에 호출합니다.")
    # Send를 사용하여 여러 전문가에게 동시에 작업을 지시합니다.
    return [
        Send(c["agent"], {"query": c["sub_query"]})
        for c in classifications
    ]
```

```py 보조
# 입구에서 Send로 나눠 보내고, 전문가 답을 모아 하나로 합치는 그래프입니다.
def synthesize(state):
    return {"final_answer": "\n".join(state["results"])}

builder = StateGraph(RouterState)
for name, role in [("electronics_agent", "전자제품"), ("clothing_agent", "의류")]:
    builder.add_node(name, make_expert(role))
    builder.add_edge(name, "synthesize")
builder.add_node("synthesize", synthesize)
builder.add_conditional_edges(START, route_query_parallel, ["electronics_agent", "clothing_agent"])
builder.add_edge("synthesize", END)
z_mall_workflow = builder.compile()

out = z_mall_workflow.invoke({"query": "여름 휴가 때 입을 원피스랑, 거기서 쓸 방수 카메라 추천해줘"})
```

## 13.6 커스텀 워크플로우

```py 원고
from typing import TypedDict

class ZMallState(TypedDict):
    question: str   # 고객의 질문
    category: str   # 분류된 의도 (배송 or 환불)
    answer: str     # 최종 답변
```

```py 원고
from langchain.chat_models import init_chat_model

llm = init_chat_model("gpt-4o-mini")

# [노드 1] 의도 분류기 (AI가 판단)
def classify_intent(state: ZMallState) -> dict:
    question = state["question"]
    prompt = f"질문: '{question}'\n이 질문이 '배송' 관련인지 '환불' 관련인지 단어 하나로만 대답해."
    response = llm.invoke(prompt)
    return {"category": response.content.strip()}

# [노드 2] 배송 조회 (결정론적 로직)
def check_delivery(state: ZMallState) -> dict:
    return {"answer": "고객님의 상품은 현재 '대전 허브'에서 이동 중입니다."}

# [노드 3] 환불 상담 (에이전트 로직)
def handle_refund(state: ZMallState) -> dict:
    question = state["question"]
    prompt = f"고객 질문: {question}\n고객이 환불을 원합니다. 규정을 설명하고 정중하게 위로하는 답변을 작성해줘."
    response = llm.invoke(prompt)
    return {"answer": response.content}
```

```py 원고
from langgraph.graph import StateGraph, START, END

workflow = StateGraph(ZMallState)

# 노드 추가
workflow.add_node("classifier", classify_intent)
workflow.add_node("delivery_service", check_delivery)
workflow.add_node("refund_agent", handle_refund)

# 시작점 연결
workflow.add_edge(START, "classifier")

# 조건부 분기
def route_decision(state: ZMallState):
    if "배송" in state["category"]:
        return "delivery_service"
    else:
        return "refund_agent"

workflow.add_conditional_edges(
    "classifier",
    route_decision,
    {
        "delivery_service": "delivery_service",
        "refund_agent": "refund_agent"
    }
)

workflow.add_edge("delivery_service", END)
workflow.add_edge("refund_agent", END)

app = workflow.compile()
```
