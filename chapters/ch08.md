### 8.1.1 에이전트의 첫 번째 구조: 반복(Loop)

```py 원고
from typing_extensions import TypedDict
from langgraph.graph import StateGraph, START, END

class State(TypedDict):
    score: int
    attempt: int

def step(state: State) -> State:
    # 무언가 시도하고 점수를 업데이트한다고 가정
    return {
        "score": state["score"] + 10,
        "attempt": state["attempt"] + 1,
    }

def should_continue(state: State) -> str:
    if state["score"] >= 80 or state["attempt"] >= 3:
        return "end"
    return "retry"

builder = StateGraph(State)
builder.add_node("step", step)

builder.add_edge(START, "step")
builder.add_conditional_edges(
    "step",
    should_continue,
    {"retry": "step", "end": END},
)

graph = builder.compile()
print(graph.invoke({"score": 50, "attempt": 0}))
```

```py 원고
from typing_extensions import TypedDict, NotRequired
from langgraph.graph import StateGraph, START, END

class SlotState(TypedDict):
    user_input: str
    budget: NotRequired[int]
    cordless: NotRequired[bool]
    question: NotRequired[str]

def extract_slots(state: SlotState) -> SlotState:
    text = state["user_input"]
    updates = {}
    if "10만" in text:
        updates["budget"] = 100000
    if "무선" in text:
        updates["cordless"] = True
    if "유선" in text:
        updates["cordless"] = False
    return updates

def ask_missing(state: SlotState) -> SlotState:
    missing = []
    if "budget" not in state:
        missing.append("예산")
    if "cordless" not in state:
        missing.append("무선/유선 선호")
    if not missing:
        return {"question": ""}
    return {"question": f"{', '.join(missing)} 정보를 알려주세요."}
```

```py 원고
builder = StateGraph(SlotState)
builder.add_node("extract", extract_slots)
builder.add_node("ask", ask_missing)
builder.add_edge(START, "extract")
builder.add_edge("extract", "ask")
builder.add_edge("ask", END)
graph = builder.compile()
```

```py 읽기전용
#! note: ipywidgets 위젯은 Jupyter 노트북에서만 동작합니다. 아래 셀이 같은 흐름을 정해 둔 답으로 실행합니다.
from ipywidgets import Text, Button, VBox, HTML, Output
from IPython.display import display

store = {"state": {"user_input": "청소기 추천해줘. 무선으로!"}, "out": None}
store["out"] = graph.invoke(store["state"])
question = store["out"].get("question") or ""

question_label = HTML(value=f"<b>봇:</b> {question or '(슬롯이 모두 채워졌습니다)'}")
reply_text = Text(placeholder="예: 10만원이요", description="입력:")
submit_btn = Button(description="제출")
result_out = Output()

def on_submit(btn):
    with result_out:
        result_out.clear_output()
        reply = (reply_text.value or "").strip()
        if not reply:
            print("입력 후 [제출]을 눌러주세요.")
            return
        store["state"] = {**store["out"], "user_input": store["out"]["user_input"] + " " + reply}
        reply_text.value = ""
        store["out"] = graph.invoke(store["state"])
        q = store["out"].get("question") or ""
        question_label.value = f"<b>봇:</b> {q or '(슬롯이 모두 채워졌습니다)'}"
        if q == "":
            print("[슬롯 채워짐] 최종 state:", store["out"])
            submit_btn.disabled = True

submit_btn.on_click(on_submit)
display(VBox([question_label, reply_text, submit_btn, result_out]))
```

### 8.1.2 에이전트의 두 번째 구조: 판단(Decision)

```py 원고
from typing_extensions import TypedDict
from langgraph.graph import StateGraph, START, END

class DecideState(TypedDict):
    query: str
    intent: str

def normalize(state: DecideState) -> DecideState:
    # 고정 절차(예: 전처리, 정규화)
    return {"query": state["query"].strip()}

def classify_intent(state: DecideState) -> DecideState:
    q = state["query"]
    if "환불" in q or "취소" in q:
        return {"intent": "refund"}
    if "추천" in q:
        return {"intent": "recommend"}
    return {"intent": "qa"}

def route(state: DecideState) -> str:
    return state["intent"]

def handle_refund(_: DecideState) -> DecideState:
    return {"intent": "refund_done"}

def handle_recommend(_: DecideState) -> DecideState:
    return {"intent": "recommend_done"}

def handle_qa(_: DecideState) -> DecideState:
    return {"intent": "qa_done"}

builder = StateGraph(DecideState)
builder.add_node("normalize", normalize)
builder.add_node("classify", classify_intent)
builder.add_node("refund", handle_refund)
builder.add_node("recommend", handle_recommend)
builder.add_node("qa", handle_qa)

builder.add_edge(START, "normalize")
builder.add_edge("normalize", "classify")
builder.add_conditional_edges(
    "classify",
    route,
    {"refund": "refund", "recommend": "recommend", "qa": "qa"},
)

builder.add_edge("refund", END)
builder.add_edge("recommend", END)
builder.add_edge("qa", END)

graph = builder.compile()
print(graph.invoke({"query": "환불 요청합니다", "intent": ""}))
```

### 8.1.3 에이전트의 세 번째 구조: 분해(Modularity)

```py 원고
from typing_extensions import TypedDict
from langgraph.graph import StateGraph, START, END

class RecommendState(TypedDict):
    query: str          # 사용자 검색어
    items: list[str]    # 추천 후보

def recommend(state: RecommendState) -> RecommendState:
    q = state["query"]
    return {"items": [f"{q}-A", f"{q}-B", f"{q}-C"]}

rec = StateGraph(RecommendState)
rec.add_node("recommend", recommend)
rec.add_edge(START, "recommend")
rec.add_edge("recommend", END)
recommend_subgraph = rec.compile()
```

```py 원고
class MainState(TypedDict):
    query: str
    items: list[str]
    final: str

def call_recommend(state: MainState) -> MainState:
    result = recommend_subgraph.invoke({"query": state["query"], "items": []})
    return {"items": result["items"]}

def finalize(state: MainState) -> MainState:
    return {"final": state["items"][0]}

main = StateGraph(MainState)
main.add_node("call_recommend", call_recommend)
main.add_node("finalize", finalize)
main.add_edge(START, "call_recommend")
main.add_edge("call_recommend", "finalize")
main.add_edge("finalize", END)
graph = main.compile()
```

```py 원고
print(graph.invoke({"query": "청소기", "items": [], "final": ""}))
```

### 8.1.4 세 구조를 하나로 합치면 “에이전트 골격”이 된다 (통합 예제)

```py 원고
from typing_extensions import TypedDict
from langgraph.graph import StateGraph, START, END

# (1) 추천 서브그래프
class RecommendState(TypedDict):
    query: str
    items: list[str]

def recommend(state: RecommendState) -> RecommendState:
    q = state["query"]
    # 단순 예시: 추천 후보 생성
    return {"items": [f"{q}-A", f"{q}-B", f"{q}-C"]}

rec = StateGraph(RecommendState)
rec.add_node("recommend", recommend)
rec.add_edge(START, "recommend")
rec.add_edge("recommend", END)
recommend_subgraph = rec.compile()

# (2) 메인 그래프 상태
class AgentState(TypedDict):
    query: str
    intent: str
    items: list[str]
    score: int
    attempt: int

def classify_intent(state: AgentState) -> AgentState:
    q = state["query"]
    if "추천" in q:
        return {"intent": "recommend"}
    return {"intent": "qa"}

def route(state: AgentState) -> str:
    return state["intent"]

def call_recommend(state: AgentState) -> AgentState:
    result = recommend_subgraph.invoke({"query": state["query"], "items": []})
    return {"items": result["items"]}

def evaluate(state: AgentState) -> AgentState:
    # 예시: attempt가 늘수록 점수가 오른다고 가정(실무에서는 품질평가/룰/LLM평가 등)
    new_attempt = state["attempt"] + 1
    new_score = state["score"] + 30
    return {"attempt": new_attempt, "score": new_score}

def should_retry(state: AgentState) -> str:
    if state["score"] >= 80 or state["attempt"] >= 3:
        return "end"
    return "retry"

def answer_qa(_: AgentState) -> AgentState:
    # QA 처리(여기서는 단순히 종료용)
    return {"score": 100}

builder = StateGraph(AgentState)
builder.add_node("classify", classify_intent)
builder.add_node("recommend", call_recommend)
builder.add_node("evaluate", evaluate)
builder.add_node("qa", answer_qa)

builder.add_edge(START, "classify")
builder.add_conditional_edges("classify", route, {"recommend": "recommend", "qa": "qa"})

# 추천 흐름: 서브그래프 호출 → 평가 → 반복/종료
builder.add_edge("recommend", "evaluate")
builder.add_conditional_edges("evaluate", should_retry, {"retry": "recommend", "end": END})

# QA는 바로 종료
builder.add_edge("qa", END)

graph = builder.compile()

result = graph.invoke({"query": "청소기 추천해줘", "intent": "", "items": [], "score": 0, "attempt": 0})
print(result)
```

### 8.2.3 그래프 API 예제: 동물 이야기 생성 (모드별 스트리밍 실습)

```py 원고
from typing import TypedDict

class StoryState(TypedDict):
    animal: str
    intro: str
    story: str
    ending: str
```

```py 원고
from langgraph.config import get_stream_writer

def uppercase_animal(state: StoryState):
    writer = get_stream_writer()
    writer("동물 이름을 대문자로 바꾸는 중입니다...")
    # 한글에는 대소문자가 없어 "토끼"는 그대로 나옵니다.
    # 영문으로 "rabbit"을 넣으면 "RABBIT"으로 바뀝니다.
    return {"animal": state["animal"].upper()}

def create_intro(state: StoryState):
    writer = get_stream_writer()
    writer("소개 문장을 만들고 있습니다...")
    return {"intro": f"한때 {state['animal']}가 있었습니다."}

def create_story(state: StoryState):
    writer = get_stream_writer()
    writer("이야기를 완성하고 있습니다...")
    return {"story": f"{state['intro']} 그 {state['animal']}는 모험을 떠났습니다."}
```

```py 원고
from langchain.chat_models import init_chat_model

model = init_chat_model(model="gpt-4o-mini")

def create_ending_with_llm(state: StoryState):
    writer = get_stream_writer()
    writer("LLM이 마지막 문장을 생성 중입니다...")

    prompt = f"다음 이야기의 마무리 문장을 한 문장으로 써줘: {state['story']}"

    # 스트리밍 대신 최종 결과만 ending에 저장합니다.
    # (토큰 스트리밍 출력은 messages stream_mode에서 관찰하게 됩니다.)
    response = model.invoke([{"role": "user", "content": prompt}])

    return {"ending": response.content}
```

```py 원고

    .add_edge(START, "uppercase_animal")
    .add_edge("uppercase_animal", "create_intro")
    .add_edge("create_intro", "create_story")
    .add_edge("create_story", "create_ending_with_llm")
    .add_edge("create_ending_with_llm", END)
    .compile()
)
```

### 8.2.4 Functional API 예제: 동물 이야기 생성(모드별 스트리밍 실습)

```py 원고
from typing import TypedDict

from langgraph.func import entrypoint, task
from langgraph.config import get_stream_writer
from langgraph.checkpoint.memory import InMemorySaver

from langchain.chat_models import init_chat_model

# LLM 준비
model = init_chat_model(model="gpt-4o-mini")

# 입력/출력 타입(선택이지만 정해 두면 읽기 쉽습니다)
class StoryInput(TypedDict):
    animal: str

class StoryOutput(TypedDict):
    animal: str
    intro: str
    story: str
    ending: str

@task
def uppercase_animal(animal: str) -> str:
    # 한글은 그대로, 영문은 대문자로 바뀝니다("rabbit" → "RABBIT").
    return animal.upper()

@task
def create_intro(animal_upper: str) -> str:
    return f"한때 {animal_upper}가 있었습니다."

@task
def create_story(intro: str, animal_upper: str) -> str:
    return f"{intro} 그 {animal_upper}는 모험을 떠났습니다."

@task
def create_ending_with_llm(story: str) -> str:
    prompt = f"다음 이야기의 마무리 문장을 한 문장으로 써줘: {story}"
    response = model.invoke([{"role": "user", "content": prompt}])
    return response.content

checkpointer = InMemorySaver()

@entrypoint(checkpointer=checkpointer)
def story_flow(inputs: StoryInput) -> StoryOutput:
    writer = get_stream_writer()

    writer("Functional API 워크플로를 시작합니다...")

    animal_upper = uppercase_animal(inputs["animal"]).result()
    writer(f"[1/4] 동물 이름 변환 완료: {animal_upper}")

    intro = create_intro(animal_upper).result()
    writer(f"[2/4] 소개 문장 생성 완료")

    story = create_story(intro, animal_upper).result()
    writer(f"[3/4] 본문 이야기 생성 완료")

    writer("[4/4] LLM이 마무리 문장을 생성 중입니다...")
    ending = create_ending_with_llm(story).result()

    writer("모든 단계 실행이 완료되었습니다.")

    return {
        "animal": animal_upper,
        "intro": intro,
        "story": story,
        "ending": ending,
    }
```
