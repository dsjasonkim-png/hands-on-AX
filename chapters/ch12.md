### 12.2.3 환경 변수 확인

```py 보조
#! note: 키 값은 보여 주지 않고 설정 여부만 출력합니다.
import os
for k in ["OPENAI_API_KEY", "LANGSMITH_TRACING", "LANGSMITH_API_KEY", "LANGSMITH_PROJECT"]:
    v = os.environ.get(k)
    shown = v if k in ("LANGSMITH_TRACING", "LANGSMITH_PROJECT") else ("설정됨" if v else None)
    print(f"{k:20} {shown or '없음'}")
```

### 12.3.2 태그·메타데이터를 붙여 실행하기

```py 보조
# 12장이 들여다볼 Z몰 고객지원 에이전트(4장·11장과 같은 모양)
from langchain_core.tools import tool
from langchain.agents import create_agent as create_react_agent

@tool
def lookup_order(order_id: str) -> str:
    """Z몰 주문 DB에서 주문번호로 주문 상태를 조회합니다."""
    return f"주문 {order_id}: 배송 중, 11/09 도착 예정"

agent = create_react_agent(
    model="gpt-4o-mini",
    tools=[lookup_order],
    system_prompt="당신은 Z몰 고객지원 상담원입니다. 도착일은 조회 결과의 날짜를 그대로 안내합니다.",
)
```

```py 원고
result = agent.invoke(
    {"messages": [{"role": "user", "content": "ZM-1029 주문 어디쯤이야?"}]},
    config={
        "tags": ["prompt-v2", "cs-chat"],
        "metadata": {"user_id": "user_1029", "segment": "VIP", "thread_id": "cs-user_1029-1108"},
    },
)
```

```py 보조
print(result["messages"][-1].content)
```

### 12.5.3 경로·승인 검사

```py 원고
def grade_control_flow(case, got):
    """got: 실행 어댑터가 만든 공통 필드. 14장 axcore.evaluate.normalize()와 같은 이름을 쓴다."""
    checks = {k: got.get(k) == v for k, v in case["expect"].items()}   # 기대 상태와 비교
    checks["approval_ok"] = not (got["committed"] and got["approval_action"] != "approve")  # 승인 없는 확정 금지
    return {
        "case_id": case["case_id"],
        "checks": checks,
        "passed_control_flow": all(checks.values()),
    }

# 문자열 응답에서 승인 여부를 추측하지 않고 그래프 상태에서 읽습니다.
# 이 검사는 사실성·문장 품질까지 판정하지 않습니다.
```

```py 보조
# 위 JSON 사례 하나와, 어댑터가 만들었다고 가정한 결과 두 가지로 채점해 봅니다.
case = {
    "case_id": "missing_offer_01",
    "expect": {"status": "needs_info", "stopped_at": "clarify"},
}
good = {"outcome": "interrupted", "status": "needs_info", "stopped_at": "clarify",
        "approval_action": None, "committed": False, "questions": ["할인율과 유효기간을 알려 주세요."]}
bad = {"outcome": "committed", "status": "completed", "stopped_at": None,
       "approval_action": "revise", "committed": True, "questions": []}

for name, got in [("되물은 경우", good), ("할인율을 지어내 확정한 경우", bad)]:
    print(name, "→", grade_control_flow(case, got))
```

평가 사례를 실제 그래프로 돌리는 전체 흐름은 14장의 공통 골격에 들어 있습니다. 14장 페이지에서 `live_check.py`를 실행해 보세요.
