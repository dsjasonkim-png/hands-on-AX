## 6.1 장기 기억의 개요와 저장 구조

### 6.1.4 기본 메모리 저장 및 검색

```py 원고
from langchain.embeddings import init_embeddings
from langgraph.store.memory import InMemoryStore

embeddings = init_embeddings("openai:text-embedding-3-small")

store = InMemoryStore(
    index={"embed": embeddings, "dims": 1536} # dims는 임베딩 벡터의 길이입니다
)
```

```py 원고
user_id = "user_1029"
namespace = (user_id, "z_mall_cs")

store.put(
    namespace,
    "user-preferences",
    {
        "rules": [
            "고객은 결론부터 짧게 듣기를 원함",
            "주로 모바일 앱을 통해 쇼핑함",
        ],
        "brand_preference": "Nike",
    },
)
```

```py 원고
item = store.get(namespace, "user-preferences")
print(item.value)
```

```py 원고
items = store.search(
    namespace,
    query="고객의 언어 습관",
    filter={"brand_preference": "Nike"},
)

for item in items:
    print(item.key, item.value)
```

## 6.2 에이전트 도구에서 장기 기억 읽기

### 6.2.2 장기 기억을 읽는 도구 만들기

```py 원고
from langgraph.store.memory import InMemoryStore

store = InMemoryStore()

store.put(
    ("users",),
    "user_1029",
    {
        "name": "김민지",
        "language": "Korean",
        "membership": "VIP",
    },
)
```

```py 원고
from typing import Annotated
from langchain_core.runnables import RunnableConfig
from langchain_core.tools import tool
from langgraph.store.base import BaseStore
from langgraph.prebuilt import InjectedStore

@tool
def get_user_info(
    config: RunnableConfig,
    store: Annotated[BaseStore, InjectedStore()],
) -> str:
    """현재 접속 중인 고객의 기본 정보(이름, 언어, 회원 등급)를 조회합니다."""
    user_id = config.get("configurable", {}).get("user_id")
    if not user_id:
        return "고객 정보를 확인할 수 없습니다."
    user_info = store.get(("users",), user_id)
    return str(user_info.value) if user_info else "등록된 고객 정보가 없습니다."
```

```py 원고
from langchain.chat_models import init_chat_model
from langchain.agents import create_agent as create_react_agent

llm = init_chat_model("gpt-4o-mini", temperature=0)

agent = create_react_agent(
    model=llm,
    tools=[get_user_info],
    store=store,
)

result = agent.invoke(
    {"messages": [{"role": "user", "content": "내 고객 정보를 조회해줘"}]},
    config={"configurable": {"user_id": "user_1029"}},
)
print(result["messages"][-1].content)
```

## 6.3 에이전트 도구에서 장기 기억 쓰기

### 6.3.2 장기 기억을 갱신하는 도구 만들기

```py 원고
from typing_extensions import TypedDict, NotRequired

class UserInfoUpdate(TypedDict):
    name: NotRequired[str]
    language: NotRequired[str]
```

```py 원고
@tool
def save_user_info(
    user_info: UserInfoUpdate,
    config: RunnableConfig,
    store: Annotated[BaseStore, InjectedStore()],
) -> str:
    """고객이 직접 바꿔 달라고 요청한 이름이나 사용 언어를 저장합니다."""
    user_id = config.get("configurable", {}).get("user_id")
    if not user_id:
        return "고객 정보를 확인할 수 없어 저장하지 않았습니다."
    current = store.get(("users",), user_id)
    merged = {**(current.value if current else {}), **user_info}
    store.put(("users",), user_id, merged)
    return "고객 정보를 저장했습니다."
```

```py 원고
agent = create_react_agent(
    model=llm,
    tools=[save_user_info],
    store=store,
)

agent.invoke(
    {"messages": [{"role": "user", "content": "앞으로는 영문 이름 John Smith로 불러 주세요."}]},
    config={"configurable": {"user_id": "user_1029"}},
)
```

```py 원고
result = store.get(("users",), "user_1029")
print(result.value)
```

### 6.3.3 읽기와 쓰기를 함께 사용하기

```py 원고
agent = create_react_agent(
    model=llm,
    tools=[get_user_info, save_user_info],
    store=store,
)

config = {"configurable": {"user_id": "user_1029"}}

# 언어 변경 요청
agent.invoke(
    {"messages": [{"role": "user", "content": "앞으로는 영어로 안내해 주세요."}]},
    config=config,
)

# 변경 후 조회
result = agent.invoke(
    {"messages": [{"role": "user", "content": "내 정보 알려줘."}]},
    config=config,
)
print(result["messages"][-1].content)
```

### 6.3.4 기억을 고치고 지우기

```py 원고
def revoke_marketing_consent(store: BaseStore, user_id: str) -> list[str]:
    """마케팅 수신 동의를 철회하고, 동의를 전제로 저장한 기억을 지웁니다."""
    namespace = (user_id, "z_mall_marketing")
    deleted = []
    for key in ["interest-items", "browsing-signals"]:
        if store.get(namespace, key):
            store.delete(namespace, key)
            deleted.append(key)
    store.put(namespace, "consent", {"marketing": False})
    return deleted

store.put(("user_1029", "z_mall_marketing"), "interest-items", {"items": ["러닝화", "바람막이"]})
store.put(("user_1029", "z_mall_marketing"), "consent", {"marketing": True})

print(revoke_marketing_consent(store, "user_1029"))
print(store.get(("user_1029", "z_mall_marketing"), "interest-items"))
print(store.get(("user_1029", "z_mall_marketing"), "consent").value)
```
