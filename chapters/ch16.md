### 16.3 주문 DB 들여다보기

```py 원고
from jobs.commerce_analysis.readonly_sql import run_readonly_sql
from jobs.commerce_analysis.spec import DEFAULT_DB

print(run_readonly_sql(DEFAULT_DB, """
    SELECT session_id, started_at, channel, viewed_product, added_to_cart, started_checkout, order_id
    FROM sessions WHERE session_id IN ('S2609-01132', 'S2609-00466', 'S2609-02598') ORDER BY started_at"""))
```

```py 원고
print(run_readonly_sql(DEFAULT_DB, """
    SELECT o.order_id, i.product_id, i.qty, i.unit_price, i.discount, p.amount
    FROM orders o JOIN order_items i ON i.order_id = o.order_id
                  JOIN payments p ON p.order_id = o.order_id
    WHERE o.order_id = 'O2609-0001'"""))
```

```py 원고
#! note: 책의 실행 결과는 openai:gpt-6-luna로 돌린 것입니다. 같은 모델로 보려면 `편집`을 눌러 DEFAULT_MODEL 자리에 "openai:gpt-6-luna"를 넣으세요. 기본 모델(DEFAULT_MODEL)은 Gemini입니다. 가짜 모델 모드에서는 model = None(가짜 판단기)으로 실행합니다.
#! fake: from axcore import Trigger, pending, resume, start; from jobs.commerce_analysis.eval_cases import make_graph; from jobs.commerce_analysis.spec import fill_decision; model = None; graph = make_graph(model=model, runtime="runtime/book_examples")
from langchain.chat_models import init_chat_model

from axcore import Trigger, pending, resume, start
from jobs import DEFAULT_MODEL
from jobs.commerce_analysis.eval_cases import make_graph
from jobs.commerce_analysis.spec import fill_decision

model = init_chat_model(DEFAULT_MODEL, temperature=0)    # API 키 없이 흐름만 보려면 model = None
graph = make_graph(model=model, runtime="runtime/book_examples")
```

### 예제 ③: 돌려보낸 뒤 담당자 승인을 기다린다

```py 원고
#! note: 실제 모델은 실행마다 결과가 다릅니다. 한 번에 통과해 승인 요청 없이 보고서가 저장되거나, 기본 모델(Gemini)처럼 같은 실수를 되풀이해 모델 오류 상한에 닿아 failed로 끝나면(반영 없음) 멈춘 요청이 없어 이 셀이 오류를 냅니다.
out = start(graph, Trigger(request_id="Q3", payload={"question": "9월에 클릭은 8%밖에 안 줄었는데 매출은 왜 18%나 줄었어?",
                                                     "as_of": "2026-10-05"}))
ask = pending(out)
shown = fill_decision(ask["decision"])      # 사람에게 보여 줄 때도 숫자는 코드가 채운다
print(ask["kind"], shown["status"], shown["primary_cause"], shown["primary_channel"])
for f in out["feedback"][:3]:
    print("피드백:", f)
print(shown["summary"])
```
